import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
from sklearn.ensemble import RandomForestClassifier
from sklearn.preprocessing import LabelEncoder
import os

# Load data
df = pd.read_csv(os.environ['INPUT_DIR'] + '/streaming.csv')

# Prepare features for modeling
df_model = df.copy()
le_tier = LabelEncoder()
le_genre = LabelEncoder()
df_model['SubscriptionTier_enc'] = le_tier.fit_transform(df_model['SubscriptionTier'])
df_model['TopGenre_enc'] = le_genre.fit_transform(df_model['TopGenre'])

# Features for modeling
feature_cols = ['SubscriptionTier_enc', 'TotalViewingHoursLastMonth', 'TopGenre_enc',
                'BingeWatchingSessionsLastMonth', 'NumberOfUniqueTitlesWatchedLastMonth',
                'AverageSessionDurationMinutes', 'CustomerServiceInteractionsLastYear', 'MonthlyCost']

X = df_model[feature_cols]
y = df_model['Churned']

# Train Random Forest
rf = RandomForestClassifier(n_estimators=200, max_depth=10, random_state=42)
rf.fit(X, y)

# Feature importance
feature_importance = pd.DataFrame({
    'feature': feature_cols,
    'importance': rf.feature_importances_
}).sort_values('importance', ascending=False)

print("Feature Importance:")
print(feature_importance)

# Analyze key drivers
print("\n" + "="*50)
print("CHURN ANALYSIS BY KEY FEATURES")
print("="*50)

# 1. Viewing Hours
print("\n1. Total Viewing Hours:")
churned_hours = df[df['Churned']==1]['TotalViewingHoursLastMonth']
retained_hours = df[df['Churned']==0]['TotalViewingHoursLastMonth']
print(f"  Churned: {churned_hours.mean():.1f} hrs (median: {churned_hours.median():.1f})")
print(f"  Retained: {retained_hours.mean():.1f} hrs (median: {retained_hours.median():.1f})")

# 2. Customer Service Interactions
print("\n2. Customer Service Interactions:")
churned_cs = df[df['Churned']==1]['CustomerServiceInteractionsLastYear']
retained_cs = df[df['Churned']==0]['CustomerServiceInteractionsLastYear']
print(f"  Churned: {churned_cs.mean():.2f} interactions (median: {churned_cs.median():.0f})")
print(f"  Retained: {retained_cs.mean():.2f} interactions (median: {retained_cs.median():.0f})")

# 3. Subscription Tier
print("\n3. Subscription Tier:")
tier_churn = df.groupby('SubscriptionTier')['Churned'].agg(['sum', 'count', 'mean'])
tier_churn.columns = ['Churned_Count', 'Total', 'Churn_Rate']
print(tier_churn)

# 4. Unique Titles
print("\n4. Unique Titles Watched:")
churned_titles = df[df['Churned']==1]['NumberOfUniqueTitlesWatchedLastMonth']
retained_titles = df[df['Churned']==0]['NumberOfUniqueTitlesWatchedLastMonth']
print(f"  Churned: {churned_titles.mean():.1f} titles (median: {churned_titles.median():.1f})")
print(f"  Retained: {retained_titles.mean():.1f} titles (median: {retained_titles.median():.1f})")

# 5. Binge Watching
print("\n5. Binge Watching Sessions:")
churned_binge = df[df['Churned']==1]['BingeWatchingSessionsLastMonth']
retained_binge = df[df['Churned']==0]['BingeWatchingSessionsLastMonth']
print(f"  Churned: {churned_binge.mean():.1f} sessions (median: {churned_binge.median():.1f})")
print(f"  Retained: {retained_binge.mean():.1f} sessions (median: {retained_binge.median():.1f})")

# Save for plotting
np.save('/tmp/feature_importance.npy', feature_importance.values)
feature_importance.to_csv('/tmp/feature_importance.csv', index=False)

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
import os

# Load data
df = pd.read_csv(os.environ['INPUT_DIR'] + '/streaming.csv')

# Set style
sns.set_style("whitegrid")
plt.rcParams['figure.facecolor'] = 'white'

# Create comprehensive dashboard
fig = plt.figure(figsize=(20, 12))
gs = fig.add_gridspec(3, 3, hspace=0.35, wspace=0.3)

# Title
fig.suptitle('Comprehensive Churn Analysis - Major Drivers', fontsize=20, fontweight='bold', y=0.98)

# 1. Feature Importance (Top left - larger)
ax1 = fig.add_subplot(gs[0, :2])
feature_imp = pd.read_csv('/tmp/feature_importance.csv')
# Rename for readability
feature_names = {
    'TotalViewingHoursLastMonth': 'Viewing Hours',
    'AverageSessionDurationMinutes': 'Avg Session Duration',
    'NumberOfUniqueTitlesWatchedLastMonth': 'Unique Titles',
    'TopGenre_enc': 'Genre Preference',
    'BingeWatchingSessionsLastMonth': 'Binge Sessions',
    'CustomerServiceInteractionsLastYear': 'CS Interactions',
    'MonthlyCost': 'Monthly Cost',
    'SubscriptionTier_enc': 'Subscription Tier'
}
feature_imp['feature'] = feature_imp['feature'].map(feature_names)
colors = ['#d73027' if x > 0.15 else '#fc8d59' if x > 0.10 else '#fee090' for x in feature_imp['importance']]
bars = ax1.barh(feature_imp['feature'], feature_imp['importance'], color=colors)
ax1.set_xlabel('Importance Score', fontsize=12, fontweight='bold')
ax1.set_title('Feature Importance (Random Forest)', fontsize=14, fontweight='bold', pad=10)
ax1.invert_yaxis()
for i, v in enumerate(feature_imp['importance']):
    ax1.text(v + 0.005, i, f'{v:.3f}', va='center', fontsize=10)

# 2. Viewing Hours Distribution
ax2 = fig.add_subplot(gs[0, 2])
churned = df[df['Churned']==1]['TotalViewingHoursLastMonth']
retained = df[df['Churned']==0]['TotalViewingHoursLastMonth']
ax2.hist([retained, churned], bins=20, label=['Retained', 'Churned'], 
         color=['#4daf4a', '#e41a1c'], alpha=0.7, edgecolor='black')
ax2.set_xlabel('Viewing Hours', fontsize=10, fontweight='bold')
ax2.set_ylabel('Count', fontsize=10, fontweight='bold')
ax2.set_title('Viewing Hours Distribution', fontsize=12, fontweight='bold')
ax2.legend()
ax2.axvline(churned.mean(), color='#e41a1c', linestyle='--', linewidth=2, label='Churned Mean')
ax2.axvline(retained.mean(), color='#4daf4a', linestyle='--', linewidth=2, label='Retained Mean')

# 3. Customer Service Interactions
ax3 = fig.add_subplot(gs[1, 0])
cs_data = df.groupby(['CustomerServiceInteractionsLastYear', 'Churned']).size().unstack(fill_value=0)
cs_pct = cs_data.div(cs_data.sum(axis=1), axis=0) * 100
cs_pct.plot(kind='bar', stacked=True, ax=ax3, color=['#4daf4a', '#e41a1c'], 
            legend=False, edgecolor='black', linewidth=0.5)
ax3.set_xlabel('CS Interactions (Last Year)', fontsize=10, fontweight='bold')
ax3.set_ylabel('Percentage (%)', fontsize=10, fontweight='bold')
ax3.set_title('Churn Rate by CS Interactions', fontsize=12, fontweight='bold')
ax3.set_xticklabels(ax3.get_xticklabels(), rotation=0)
ax3.legend(['Retained', 'Churned'], loc='upper left')

# 4. Subscription Tier
ax4 = fig.add_subplot(gs[1, 1])
tier_data = df.groupby('SubscriptionTier')['Churned'].agg(['sum', 'count'])
tier_data['retained'] = tier_data['count'] - tier_data['sum']
tier_data[['retained', 'sum']].plot(kind='bar', ax=ax4, color=['#4daf4a', '#e41a1c'],
                                     edgecolor='black', linewidth=1)
ax4.set_xlabel('Subscription Tier', fontsize=10, fontweight='bold')
ax4.set_ylabel('Count', fontsize=10, fontweight='bold')
ax4.set_title('Churn by Subscription Tier', fontsize=12, fontweight='bold')
ax4.set_xticklabels(ax4.get_xticklabels(), rotation=0)
ax4.legend(['Retained', 'Churned'])
# Add churn rate on top
for i, (idx, row) in enumerate(tier_data.iterrows()):
    churn_rate = row['sum'] / row['count'] * 100
    ax4.text(i, row['count'] + 5, f'{churn_rate:.1f}%', ha='center', fontweight='bold')

# 5. Unique Titles Watched
ax5 = fig.add_subplot(gs[1, 2])
titles_churned = df[df['Churned']==1]['NumberOfUniqueTitlesWatchedLastMonth']
titles_retained = df[df['Churned']==0]['NumberOfUniqueTitlesWatchedLastMonth']
bp = ax5.boxplot([titles_retained, titles_churned], labels=['Retained', 'Churned'],
                  patch_artist=True, widths=0.6)
bp['boxes'][0].set_facecolor('#4daf4a')
bp['boxes'][1].set_facecolor('#e41a1c')
for box in bp['boxes']:
    box.set_alpha(0.7)
ax5.set_ylabel('Unique Titles', fontsize=10, fontweight='bold')
ax5.set_title('Unique Titles Watched', fontsize=12, fontweight='bold')
ax5.grid(axis='y', alpha=0.3)

# 6. Binge Watching Sessions
ax6 = fig.add_subplot(gs[2, 0])
binge_churned = df[df['Churned']==1]['BingeWatchingSessionsLastMonth']
binge_retained = df[df['Churned']==0]['BingeWatchingSessionsLastMonth']
bp2 = ax6.boxplot([binge_retained, binge_churned], labels=['Retained', 'Churned'],
                   patch_artist=True, widths=0.6)
bp2['boxes'][0].set_facecolor('#4daf4a')

# 7. Session Duration
ax7 = fig.add_subplot(gs[2, 1])
duration_churned = df[df['Churned']==1]['AverageSessionDurationMinutes']

bp3 = ax7.boxplot([duration_retained, duration_churned], labels=['Retained', 'Churned'],

bp3['boxes'][0].set_facecolor('#4daf4a')

# 8. Key Metrics Summary
ax8 = fig.add_subplot(gs[2, 2])
ax8.axis('off')

# Calculate metrics
metrics = {
    'Overall Churn Rate': f"{df['Churned'].mean()*100:.1f}%",
    'Avg Viewing (Churned)': f"{df[df['Churned']==1]['TotalViewingHoursLastMonth'].mean():.1f} hrs",
    'Avg Viewing (Retained)': f"{df[df['Churned']==0]['TotalViewingHoursLastMonth'].mean():.1f} hrs",

y_pos = 0.95
ax8.text(0.5, y_pos, 'KEY FINDINGS', ha='center', fontsize=14, fontweight='bold',
